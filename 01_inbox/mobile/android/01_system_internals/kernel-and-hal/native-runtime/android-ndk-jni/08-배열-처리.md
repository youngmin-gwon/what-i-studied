# 배열 처리

상위 노트: [android-ndk-jni](01_inbox/mobile/android/01_system_internals/kernel-and-hal/native-runtime/android-ndk-jni.md)

```cpp
extern "C" JNIEXPORT jintArray JNICALL
Java_com_example_app_NativeLib_processArray(
    JNIEnv* env,
    jobject /* this */,
    jintArray input) {
    
    // 배열 길이
    jsize length = env->GetArrayLength(input);
    
    // Java 배열 → C 배열
    jint* nativeArray = env->GetIntArrayElements(input, nullptr);
    
    // 처리
    for (int i = 0; i < length; i++) {
        nativeArray[i] *= 2;
    }
    
    // C 배열 → Java 배열
    jintArray result = env->NewIntArray(length);
    env->SetIntArrayRegion(result, 0, length, nativeArray);
    
    // 메모리 해제
    env->ReleaseIntArrayElements(input, nativeArray, 0);
    
    return result;
}
```
